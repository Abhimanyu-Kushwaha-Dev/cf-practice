for i in range(int(input())):
    n,k=map(int,input().split())
    lst=sorted(list(map(int,input().split())))
    l,r=0,n-1
    c=0
    while l<r:
        s=lst[l]+lst[r]
        if s==k:
            c+=1
            l+=1
            r-=1
        elif s<k:
            l+=1
        else:
            r-=1
    print(c)

t=int(input())
L=[]
for i in range(t):
    n,k=map(int,input().split())
    lst=sorted(map(int,input().split()))
    result=[[lst[0]]]
    for i in range(1, n):
        if lst[i]-lst[i-1]<=k:
            result[-1].append(lst[i])
        else:
            result.append([lst[i]])
    M=max(len(x) for x in result)
    L.append(n-M)
for i in L:
    print(i)

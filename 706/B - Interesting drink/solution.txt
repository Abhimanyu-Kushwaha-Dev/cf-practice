n=int(input())
prices=sorted(map(int, input().split()))
q=int(input())
for i in range(q):
    budget=int(input())
    left=0
    right=n
    while left < right:
        mid=(left + right) // 2
        if prices[mid] <= budget:
            left=mid + 1
        else:
            right=mid
    print(left)
